    <script>
    (function(){
      var vlb = document.getElementById('vid-lb');
      var frame = vlb && vlb.querySelector('.vid-lb-frame');
      var closeBtn = document.getElementById('vid-lb-close');
      if(!vlb || !frame) return;

      function openVid(id){
        frame.innerHTML = '<iframe src="https://www.youtube.com/embed/' + id +
          '?autoplay=1&rel=0&playsinline=1" title="Video testimonial"' +
          ' allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture" allowfullscreen></iframe>';
        vlb.classList.add('open');
        vlb.setAttribute('aria-hidden','false');
        document.body.style.overflow = 'hidden';
      }
      function closeVid(){
        vlb.classList.remove('open');
        vlb.setAttribute('aria-hidden','true');
        frame.innerHTML = '';
        document.body.style.overflow = '';
      }

      document.querySelectorAll('#showcase .result-card.video').forEach(function(card){
        function play(){ openVid(card.getAttribute('data-yt')); }
        card.addEventListener('click', play);
        card.addEventListener('keydown', function(e){ if(e.key==='Enter'||e.key===' '){ e.preventDefault(); play(); } });
      });

      vlb.addEventListener('click', function(e){ if(e.target === vlb) closeVid(); });
      closeBtn.addEventListener('click', closeVid);
      document.addEventListener('keydown', function(e){
        if(e.key === 'Escape' && vlb.classList.contains('open')) closeVid();
      });
    })();
    </script>

  const need = () => track.clientWidth * 2.6;

  while (count < MIN || total < need()){
    const cl = base.cloneNode(true);
    track.appendChild(cl);
    total += cl.scrollWidth + GAP;
    count++;
  }

  let x = 0, last = performance.now(), paused = false;
  const segW = () => base.scrollWidth + GAP;

  const apply = () => {
    const w = segW();
    while (-x >= w) x += w;
    while (x > 0) x -= w;
    track.style.transform = `translate3d(${x}px,0,0)`;
  };

  /* Pause on hover so reviews can actually be read */
  track.addEventListener('mouseenter', () => { paused = true; });
  track.addEventListener('mouseleave', () => { paused = false; });

  /* Touch: drag to scrub, resumes shortly after release */
  let dragX = null, resumeTimer = null;
  track.addEventListener('touchstart', (e) => {
    paused = true; clearTimeout(resumeTimer);
    dragX = e.touches[0].clientX;
  }, {passive:true});
  track.addEventListener('touchmove', (e) => {
    if (dragX === null) return;
    x += e.touches[0].clientX - dragX;
    dragX = e.touches[0].clientX;
    apply();
  }, {passive:true});
  track.addEventListener('touchend', () => {
    dragX = null;
    resumeTimer = setTimeout(() => { paused = false; }, 1600);
  });

  function step(now){
    const dt = (now - last) / 1000; last = now;
    if (!paused){
      x -= SPEED * dt;
      apply();
    }
    requestAnimationFrame(step);
  }
  requestAnimationFrame(step);

  track.dataset.marqInit = '1';
}

marquee(document.querySelector('#reviews .loop'), { speed: 24, gap: 18, minSegs: 3 });
