/* Reviews marquee */
(function(){
  function marquee(track, opts){
    if(!track || track.dataset.marqInit) return;
    var GAP = opts.gap || 18;
    var SPEED = opts.speed || 16;
    var MIN = opts.minSegs || 3;
    var original = Array.prototype.slice.call(track.children);
    track.innerHTML = '';
    function makeSeg(){
      var seg = document.createElement('div'); seg.className = 'seg';
      original.forEach(function(c){ seg.appendChild(c.cloneNode(true)); });
      return seg;
    }
    var base = makeSeg(); track.appendChild(base);
    var total = base.scrollWidth + GAP, count = 1;
    function need(){ return track.clientWidth * 2.6; }
    while(count < MIN || total < need()){
      var cl = base.cloneNode(true);
      track.appendChild(cl);
      total += cl.scrollWidth + GAP;
      count++;
    }
    var x = 0, last = performance.now(), paused = false;
    function segW(){ return base.scrollWidth + GAP; }
    function apply(){
      var w = segW();
      while(-x >= w) x += w;
      while(x > 0) x -= w;
      track.style.transform = 'translate3d(' + x + 'px,0,0)';
    }
    /* Pause on hover so reviews can actually be read */
    track.addEventListener('mouseenter', function(){ paused = true; });
    track.addEventListener('mouseleave', function(){ paused = false; });
    /* Touch: drag to scrub, resumes shortly after release */
    var dragX = null, resumeTimer = null;
    track.addEventListener('touchstart', function(e){
      paused = true; clearTimeout(resumeTimer);
      dragX = e.touches[0].clientX;
    }, {passive:true});
    track.addEventListener('touchmove', function(e){
      if (dragX === null) return;
      x += e.touches[0].clientX - dragX;
      dragX = e.touches[0].clientX;
      apply();
    }, {passive:true});
    track.addEventListener('touchend', function(){
      dragX = null;
      resumeTimer = setTimeout(function(){ paused = false; }, 1600);
    });
    function step(now){
      var dt = (now - last) / 1000; last = now;
      if (!paused){
        x -= SPEED * dt;
        apply();
      }
      requestAnimationFrame(step);
    }
    requestAnimationFrame(step);
    track.dataset.marqInit = '1';
  }
  window.addEventListener('load', function(){
    marquee(document.querySelector('#wb-reviews .loop'), { speed:24, gap:18, minSegs:3 });
  });
})();
</script>
